
% Question 13
\question
There are two solid metal spheres. The ratio of the surface area of the smaller sphere to the surface area of the larger sphere is $4 : 9$. The radius of the larger sphere is $9\text{ cm}$.
\begin{parts}
\part Express, in terms of $\pi$, the volume of the smaller sphere. \rmarks{3}
\part The two spheres are melted and recast into two solid right circular cones. Denote these two circular cones by $A$ and $B$. It is given that the height and the base radius of $A$ are $10\text{ cm}$ and $6\text{ cm}$ respectively. A student finds that the base radius of $B$ is $12\text{ cm}$. The student claims that $A$ and $B$ are similar. Is the claim correct? Explain your answer. \rmarks{4}
\end{parts}

\begin{solution}
\begin{parts}
\part The ratio of the volume of the smaller sphere to the volume of the larger sphere is $8 : 27$. \hfill (1M) \\
The volume of the smaller sphere is
\begin{align*}
&= \frac{4}{3}\pi(9)^3\left(\frac{8}{27}\right) \quad &\text{(1M)} \\
&= 288\pi\text{ cm}^3 \quad &\text{(1A)}
\end{align*}
\part The volume of $A$ is
\[
\frac{1}{3}\pi(6^2)(10) = 120\pi\text{ cm}^3 \quad \text{(1M)}
\]
The volume of $B$ is
\[
288\pi + \frac{4}{3}\pi(9)^3 - 120\pi = 1140\pi\text{ cm}^3 \quad \text{(1M)}
\]
Note that
\[
\frac{\text{The volume of } B}{\text{The volume of } A} = \frac{1140\pi}{120\pi} = \frac{19}{2} \quad \text{and} \quad \left(\frac{\text{The base radius of } B}{\text{The base radius of } A}\right)^3 = \left(\frac{12}{6}\right)^3 = 8 \ne \frac{19}{2} \quad \text{(1M)}
\]
So, $A$ and $B$ are not similar. \\
Thus, the claim is not correct. \hfill (1A f.t.)
\end{parts}
\end{solution}
