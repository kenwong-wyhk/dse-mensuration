
% Question 14
\question
A right circular cylindrical container of base radius $8\text{ cm}$ and height $64\text{ cm}$ and an inverted right circular conical vessel of base radius $20\text{ cm}$ and height $60\text{ cm}$ are held vertically. The container is fully filled with water. The water in the container is now poured into the vessel.
\begin{parts}
\part Find the volume of water in the vessel in terms of $\pi$. \rmarks{2}
\part Find the depth of water in the vessel. \rmarks{4}
\part If a solid metal sphere of radius $14\text{ cm}$ is then put into the vessel and the sphere is totally immersed in the water, will the water overflow? Explain your answer. \rmarks{3}
\end{parts}

\begin{solution}
\begin{parts}
\part The volume of water in the vessel is
\begin{align*}
&= \pi(8^2)(64) \quad &\text{(1M)} \\
&= 4096\pi\text{ cm}^3 \quad &\text{(1A)}
\end{align*}
\part Let $h\text{ cm}$ be the depth of water in the vessel. \\
Then, the radius of the water surface is $\dfrac{h}{3}\text{ cm}$. \hfill (1M)
\begin{align*}
\frac{1}{3}\pi\left(\frac{h}{3}\right)^2 h &= 4096\pi \quad &\text{(1M + 1A)} \\
h^3 &= 110592 \\
h &= 48 \quad &\text{(1A)}
\end{align*}
Thus, the depth of water in the vessel is $48\text{ cm}$.
\part The volume not occupied by water in the vessel is
\[
\frac{1}{3}\pi(20^2)(60) - 4096\pi = 3904\pi\text{ cm}^3 \quad \text{(1M)}
\]
The volume of the metal sphere is
\begin{align*}
&= \frac{4}{3}\pi(14^3) \quad &\text{(1M)} \\
&= \frac{10976}{3}\pi\text{ cm}^3 < 3904\pi\text{ cm}^3
\end{align*}
Thus, the water will not overflow. \hfill (1A f.t.)
\end{parts}
\end{solution}
