
% Question 13
\question
The base of a solid right pyramid is a square of side $64\text{ cm}$. The height of the pyramid is $24\text{ cm}$. The pyramid is divided into a frustum $X$ and a pyramid $Y$ by a plane which is parallel to its base. It is given that the height of $Y$ is $18\text{ cm}$.
\begin{parts}
\part Find the volume of $X$. \rmarks{3}
\part The base of another solid right pyramid is a square. This pyramid is divided into a frustum $Z$ and a pyramid by a plane which is parallel to its base. The height and the total surface area of $Z$ are $3\text{ cm}$ and $960\text{ cm}^2$ respectively. Are $X$ and $Z$ similar? Explain your answer. \rmarks{4}
\end{parts}

\begin{solution}
\begin{parts}
\part The volume of $X$ is
\begin{align*}
&= \frac{1}{3}(64^2)(24)\left(1 - \left(\frac{18}{24}\right)^3\right) \quad &\text{(1M + 1M)} \\
&= 18944\text{ cm}^3 \quad &\text{(1A)}
\end{align*}
\part The area of each lateral face of $X$ is
\[
\frac{1}{2}\left(64 + \frac{3}{4}(64)\right)\sqrt{6^2 + 8^2} = 560\text{ cm}^2 \quad \text{(1M)}
\]
The total surface area of $X$ is
\[
4(560) + 64^2\left(1 + \left(\frac{3}{4}\right)^2\right) = 8640\text{ cm}^2 \quad \text{(1M)}
\]
Note that
\[
\left(\frac{\text{The height of } X}{\text{The height of } Z}\right)^2 = \left(\frac{6}{3}\right)^2 = 4 \quad \text{and} \quad \frac{\text{The total surface area of } X}{\text{The total surface area of } Z} = \frac{8640}{960} = 9
\]
Since
\[
\frac{\text{The total surface area of } X}{\text{The total surface area of } Z} \ne \left(\frac{\text{The height of } X}{\text{The height of } Z}\right)^2 \quad \text{(1M)}
\]
$X$ and $Z$ are not similar. \hfill (1A f.t.)
\end{parts}
\end{solution}
