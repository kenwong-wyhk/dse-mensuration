
% Question 13
\question
A solid metal sphere of radius $10\text{ cm}$ is melted and recast into three similar solid right circular cylinders $X$, $Y$ and $Z$. Let $r_1\text{ cm}$, $r_2\text{ cm}$ and $r_3\text{ cm}$ be the base radii of $X$, $Y$ and $Z$ respectively. It is given that $r_1 : r_2 : r_3 = 2 : 3 : 5$.
\begin{parts}
\part Express the volume of $Y$ in terms of $\pi$. \rmarks{3}
\part A craftsman finds that the height of $Y$ is $9\text{ cm}$.
\begin{subparts}
\subpart Find $r_2$.
\subpart The craftsman claims that the total surface area of $Z$ is greater than the surface area of the sphere. Do you agree? Explain your answer. \\*
\null\rmarks{5}
\end{subparts}
\end{parts}

\begin{solution}
\begin{parts}
\part The volume of the solid metal sphere is
\[
\frac{4}{3}\pi (10)^3 = \frac{4000}{3}\pi\text{ cm}^3 \quad \text{(1M)}
\]
Since $X$, $Y$ and $Z$ are similar cylinders with $r_1 : r_2 : r_3 = 2 : 3 : 5$, the ratio of their volumes is $2^3 : 3^3 : 5^3 = 8 : 27 : 125$. \\
Thus, the volume of $Y$ is
\[
\frac{4000}{3}\pi \times \frac{27}{8 + 27 + 125} = 225\pi\text{ cm}^3 \quad \text{(1M + 1A)}
\]
\part
\begin{subparts}
\subpart Since the height of $Y$ is $9\text{ cm}$, we have
\begin{align*}
\pi r_2^2 (9) &= 225\pi \quad &\text{(1M)} \\
r_2^2 &= 25 \\
r_2 &= 5 \quad &\text{(1A)}
\end{align*}
\subpart The total surface area of $Y$ is
\[
2\pi (5)^2 + 2\pi (5)(9) = 140\pi\text{ cm}^2
\]
Since $Y$ and $Z$ are similar with $r_2 : r_3 = 3 : 5$, the total surface area of $Z$ is
\[
140\pi \times \left(\frac{5}{3}\right)^2 = \frac{3500}{9}\pi\text{ cm}^2 \approx 388.9\pi\text{ cm}^2 \quad \text{(1M)}
\]
The surface area of the sphere is
\[
4\pi (10)^2 = 400\pi\text{ cm}^2 > \frac{3500}{9}\pi\text{ cm}^2 \quad \text{(1M)}
\]
Thus, the claim is disagreed. \hfill (1A f.t.)
\end{subparts}
\end{parts}
\end{solution}
