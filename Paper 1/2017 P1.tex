
% Question 12
\question
A solid metal right prism of base area $84\text{ cm}^2$ and height $20\text{ cm}$ is melted and recast into two similar solid right pyramids. The bases of the two pyramids are squares. The ratio of the base area of the smaller pyramid to the base area of the larger pyramid is $4 : 9$.
\begin{parts}
\part Find the volume of the larger pyramid. \rmarks{3}
\part If the height of the larger pyramid is $12\text{ cm}$, find the total surface area of the smaller pyramid. \\*
\null\rmarks{4}
\end{parts}

\begin{solution}
\begin{parts}
\part Let $V\text{ cm}^3$ be the volume of the larger pyramid.
\begin{align*}
V + \left(\sqrt{\frac{4}{9}}\right)^3 V &= (84)(20) \quad &\text{(1M + 1M)} \\
\frac{35V}{27} &= 1680 \\
V &= 1296 \quad &\text{(1A)}
\end{align*}
Thus, the volume of the larger pyramid is $1296\text{ cm}^3$.
\part The length of the side of the base of the larger pyramid is
\[
\sqrt{\frac{3(1296)}{12}} = 18\text{ cm} \quad \text{(1M)}
\]
The total surface area of the larger pyramid is
\[
(18)(18) + 4\left(\frac{1}{2}(18)\sqrt{\left(\frac{18}{2}\right)^2 + 12^2}\right) = 864\text{ cm}^2 \quad \text{(1M)}
\]
The total surface area of the smaller pyramid is
\begin{align*}
&= \frac{4}{9}(864) \quad &\text{(1M)} \\
&= 384\text{ cm}^2 \quad &\text{(1A)}
\end{align*}
Thus, the total surface area of the smaller pyramid is $384\text{ cm}^2$.
\end{parts}
\end{solution}
