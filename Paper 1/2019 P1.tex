
% Question 9
\question
The sum of the volumes of two spheres is $324\pi\text{ cm}^3$. The radius of the larger sphere is equal to the diameter of the smaller sphere. Express, in terms of $\pi$,
\begin{parts}
\part the volume of the larger sphere;
\part the sum of the surface areas of the two spheres. \\*
\null\rmarks{5}
\end{parts}

\begin{solution}
\begin{parts}
\part Note that the ratio of the radius of the larger sphere to the radius of the smaller sphere is $2 : 1$. \\
So, the ratio of the volume of the larger sphere to the volume of the smaller sphere is $8 : 1$. \hfill (1M) \\
The volume of the larger sphere is
\[
324\pi\left(\frac{8}{1 + 8}\right) = 288\pi\text{ cm}^3 \quad \text{(1A)}
\]
\part Let $R\text{ cm}$ be the radius of the larger sphere.
\begin{align*}
\frac{4}{3}\pi R^3 &= 288\pi \quad &\text{(1M)} \\
R &= 6
\end{align*}
So, the radius of the smaller sphere is $3\text{ cm}$. \\
The sum of the surface areas of the two spheres is
\begin{align*}
&= 4\pi(6^2) + 4\pi(3^2) \quad &\text{(1M)} \\
&= 180\pi\text{ cm}^2 \quad &\text{(1A)}
\end{align*}
\end{parts}
\end{solution}
