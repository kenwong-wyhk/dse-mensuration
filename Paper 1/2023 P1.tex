
% Question 14
\question
The base radius and the curved surface area of a solid metal right circular cone are $14\text{ cm}$ and $700\pi\text{ cm}^2$ respectively.
\begin{parts}
\part Find the height of the circular cone. \rmarks{3}
\part The circular cone is divided into a right circular cone $X$ and a frustum $Y$ by a plane which is parallel to its base. The curved surface area of $Y$ is $15$ times the curved surface area of $X$.
\begin{subparts}
\subpart Express the volume of $Y$ in terms of $\pi$.
\subpart If $Y$ is melted and recast into $2$ identical solid spheres, find the diameter of each sphere. \\*
\null\rmarks{5}
\end{subparts}
\end{parts}

\begin{solution}
\begin{parts}
\part Let $l\text{ cm}$ be the slant height of the circular cone.
\begin{align*}
\pi(14)(l) &= 700\pi \quad &\text{(1M)} \\
l &= 50
\end{align*}
The height of the circular cone is
\begin{align*}
&= \sqrt{l^2 - 14^2} \quad &\text{(1M)} \\
&= \sqrt{50^2 - 14^2} \\
&= 48\text{ cm} \quad &\text{(1A)}
\end{align*}
\part
\begin{subparts}
\subpart The volume of $Y$ is
\begin{align*}
&= \frac{1}{3}\pi(14^2)(48)\left(1 - \left(\frac{1}{\sqrt{15 + 1}}\right)^3\right) \quad &\text{(1M + 1M)} \\
&= 3087\pi\text{ cm}^3 \quad &\text{(1A)}
\end{align*}
\subpart Let $d\text{ cm}$ be the diameter of each sphere.
\begin{align*}
\frac{4}{3}\pi\left(\frac{d}{2}\right)^3 &= \frac{3087\pi}{2} \quad &\text{(1M)} \\
d &= 21 \quad &\text{(1A)}
\end{align*}
Thus, the diameter of each sphere is $21\text{ cm}$.
\end{subparts}
\end{parts}
\end{solution}
