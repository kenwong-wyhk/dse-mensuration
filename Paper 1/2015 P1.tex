
% Question 9
\question
The radius and the area of a sector are $12\text{ cm}$ and $30\pi\text{ cm}^2$ respectively.
\begin{parts}
\part Find the angle of the sector.
\part Express the perimeter of the sector in terms of $\pi$. \\*
\null\rmarks{5}
\end{parts}

\begin{solution}
\begin{parts}
\part Let $\theta^\circ$ be the angle of the sector.
\begin{align*}
\frac{\theta}{360}\left(\pi(12^2)\right) &= 30\pi \quad &\text{(1M)} \\
\theta &= 75 \quad &\text{(1A)}
\end{align*}
Thus, the angle of the sector is $75^\circ$.
\part The required perimeter is
\begin{align*}
&= \frac{75}{360}(2\pi(12)) + 2(12) \quad &\text{(1M + 1M)} \\
&= (5\pi + 24)\text{ cm} \quad &\text{(1A)}
\end{align*}
\end{parts}
\end{solution}
