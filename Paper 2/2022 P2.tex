
% Q15
\question The radius of a solid hemisphere and the base radius of a solid right circular cylinder are equal. If the height of the circular cylinder is equal to its base diameter, then the ratio of the total surface area of the hemisphere to the total surface area of the circular cylinder is
\begin{choices}
    \correctchoice $1 : 2$ .
    \choice $1 : 3$ .
    \choice $2 : 3$ .
    \choice $2 : 5$ .
\end{choices}
\begin{solution}
Let $r$ be the radius of the hemisphere and the base radius of the cylinder. Then the height of the cylinder is $h = 2r$. \\
Total surface area of the hemisphere $= 3\pi r^2$. \\
Total surface area of the cylinder $= 2\pi r^2 + 2\pi r(2r) = 6\pi r^2$. \\
Required ratio $= 3\pi r^2 : 6\pi r^2 = 1 : 2$ (A).
\end{solution}

% Q16
\question The diameter of a circle is $10\text{ cm}$ . The circle is divided into a major segment and a minor segment by a chord of length $8\text{ cm}$ . Find the area of the major segment correct to the nearest $\text{cm}^2$ .
\begin{choices}
    \choice $11\text{ cm}^2$
    \choice $23\text{ cm}^2$
    \choice $55\text{ cm}^2$
    \correctchoice $67\text{ cm}^2$
\end{choices}
\begin{solution}
The radius of the circle is $r = 5\text{ cm}$. The distance from the centre $O$ to the chord $AB$ (of length $8\text{ cm}$) is $\sqrt{5^2 - 4^2} = 3\text{ cm}$, and $\angle AOB = 2\arcsin\left(\frac{4}{5}\right) \approx 106.26^\circ$. \\
Area of the major segment:
\[ \pi(5)^2 \times \frac{360^\circ - 106.26^\circ}{360^\circ} + \frac{1}{2}(8)(3) \approx 55.36 + 12 = 67\text{ cm}^2 \quad \text{(D)} \]
\end{solution}

% Q17
\question In the figure, $M$ and $N$ are points lying on $PQ$ and $QR$ respectively such that $PM : MQ = 5 : 6$ and $QN : NR = 3 : 4$ . If the area of the quadrilateral $MNRP$ is $59\text{ cm}^2$ , then the area of $\triangle MNQ$ is

\begin{minipage}{0.45\textwidth}
\begin{choices}
    \choice $17\text{ cm}^2$ .
    \correctchoice $18\text{ cm}^2$ .
    \choice $19\text{ cm}^2$ .
    \choice $20\text{ cm}^2$ .
\end{choices}
\end{minipage}
\begin{minipage}{0.52\textwidth}
\centering
\begin{tikzpicture}[scale=0.8]
    \coordinate (Q) at (0,0);
    \coordinate (R) at (5.6,0);
    \coordinate (P) at (-0.6,4.4);
    \coordinate (M) at ($(Q)!0.545!(P)$);
    \coordinate (N) at ($(Q)!0.429!(R)$);
    \draw[thick] (P) -- (Q) -- (R) -- cycle;
    \draw[thick] (M) -- (N);
    \node[above left] at (P) {$P$};
    \node[left] at (M) {$M$};
    \node[below left] at (Q) {$Q$};
    \node[below] at (N) {$N$};
    \node[right] at (R) {$R$};
\end{tikzpicture}
\end{minipage}
\begin{solution}
\[ \frac{\text{Area}(\triangle MNQ)}{\text{Area}(\triangle PQR)} = \frac{MQ}{PQ} \times \frac{QN}{QR} = \frac{6}{11} \times \frac{3}{7} = \frac{18}{77} \]
Thus $\text{Area}(\triangle MNQ) : \text{Area}(MNRP) = 18 : (77 - 18) = 18 : 59$, which gives $\text{Area}(\triangle MNQ) = 18\text{ cm}^2$ (B).
\end{solution}
