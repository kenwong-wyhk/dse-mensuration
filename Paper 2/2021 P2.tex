
% Q15
\question The base of a solid right prism is a regular $6$-sided polygon of side $8\text{ cm}$ . If the volume of the prism is $288\text{ cm}^3$ , find the total surface area of the prism correct to the nearest $\text{cm}^2$ .
\begin{choices}
    \choice $166\text{ cm}^2$
    \choice $249\text{ cm}^2$
    \correctchoice $416\text{ cm}^2$
    \choice $748\text{ cm}^2$
\end{choices}
\begin{solution}
The area of the regular hexagonal base is $6 \times \frac{\sqrt{3}}{4}(8)^2 = 96\sqrt{3}\text{ cm}^2$. \\
The height of the prism is $h = \dfrac{288}{96\sqrt{3}} = \sqrt{3}\text{ cm}$. \\
Total surface area $= 2(96\sqrt{3}) + 6(8)(\sqrt{3}) = 240\sqrt{3} \approx 416\text{ cm}^2$ (C).
\end{solution}

% Q16
\question The sum of the total surface areas of two solid hemispheres is $351\pi\text{ cm}^2$ . If the ratio of the radius of the smaller hemisphere to the radius of the larger hemisphere is $2 : 3$ , then the difference of the volumes of the two hemispheres is
\begin{choices}
    \correctchoice $342\pi\text{ cm}^3$ .
    \choice $630\pi\text{ cm}^3$ .
    \choice $684\pi\text{ cm}^3$ .
    \choice $1\,260\pi\text{ cm}^3$ .
\end{choices}
\begin{solution}
Let the radii of the two hemispheres be $2k\text{ cm}$ and $3k\text{ cm}$.
\[ 3\pi(2k)^2 + 3\pi(3k)^2 = 351\pi \implies 39k^2 = 351 \implies k^2 = 9 \implies k = 3 \]
Thus the radii are $6\text{ cm}$ and $9\text{ cm}$. The difference of their volumes is:
\[ \frac{2}{3}\pi(9^3 - 6^3) = \frac{2}{3}\pi(729 - 216) = 342\pi\text{ cm}^3 \quad \text{(A)} \]
\end{solution}

% Q17
\question The area of the sector $OAB$ is $\pi\text{ cm}^2$ , where $O$ is the centre of the sector $OAB$ . If $\angle AOB = 90^\circ$ , which of the following are true?
\begin{itemize}
    \item[I.] The radius of the sector $OAB$ is $2\text{ cm}$ .
    \item[II.] The perimeter of the sector $OAB$ is $\pi\text{ cm}$ .
    \item[III.] The area of the circle passing through $O$ , $A$ and $B$ is $2\pi\text{ cm}^2$ .
\end{itemize}
\begin{choices}
    \choice I and II only
    \correctchoice I and III only
    \choice II and III only
    \choice I, II and III
\end{choices}
\begin{solution}
\begin{itemize}
    \item I is true: $\pi r^2 \times \frac{90^\circ}{360^\circ} = \pi \implies r = 2\text{ cm}$.
    \item II is false: The perimeter of sector $OAB$ is $2(2) + 2\pi(2)\times\frac{1}{4} = (4 + \pi)\text{ cm} \ne \pi\text{ cm}$.
    \item III is true: Since $\angle AOB = 90^\circ$, $AB = \sqrt{2^2 + 2^2} = 2\sqrt{2}\text{ cm}$ is a diameter of the circle passing through $O, A, B$, so its area is $\pi(\sqrt{2})^2 = 2\pi\text{ cm}^2$.
\end{itemize}
Thus, the answer is B.
\end{solution}
