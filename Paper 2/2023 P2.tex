
% Q15
\question The length of a side of a solid cube is $60\text{ cm}$ . The volume of a solid right circular cylinder is equal to the volume of the cube while the curved surface area of the circular cylinder is equal to the total surface area of the cube. Find the base radius of the circular cylinder.
\begin{choices}
    \correctchoice $20\text{ cm}$
    \choice $30\text{ cm}$
    \choice $76\text{ cm}$
    \choice $172\text{ cm}$
\end{choices}
\begin{solution}
Let $r\text{ cm}$ and $h\text{ cm}$ be the base radius and the height of the cylinder respectively.
\[ \begin{cases} \pi r^2 h = 60^3 \\ 2\pi r h = 6(60)^2 \end{cases} \implies \frac{\pi r^2 h}{2\pi r h} = \frac{60^3}{6(60)^2} \implies \frac{r}{2} = 10 \implies r = 20\text{ cm} \quad \text{(A)} \]
\end{solution}

% Q16
\question In the figure, $AC$ is a diameter of the circle $ABCD$ while $BD$ and $EF$ are diameters of the circle $BEDF$ . It is given that $C$ and $E$ lie on $AF$ . Let $G$ be the point of intersection of $AF$ and $BD$ . If $AG = 30\text{ cm}$ and $CG = 10\text{ cm}$ , find the area of the shaded region correct to the nearest $\text{cm}^2$ .

\begin{minipage}{0.45\textwidth}
\begin{choices}
    \choice $209\text{ cm}^2$
    \choice $367\text{ cm}^2$
    \choice $383\text{ cm}^2$
    \correctchoice $540\text{ cm}^2$
\end{choices}
\end{minipage}
\begin{minipage}{0.52\textwidth}
\centering
\begin{tikzpicture}[scale=0.1]
    \coordinate (O1) at (0,0);
    \coordinate (G) at (10,0);
    \coordinate (A) at (-20,0);
    \coordinate (C) at (20,0);
    \coordinate (D) at (10,17.32);
    \coordinate (B) at (10,-17.32);
    \coordinate (E) at (-7.32,0);
    \coordinate (F) at (27.32,0);
    \begin{scope}
        \path[fill=gray!40, postaction={pattern=north east lines}] (D) arc (60:300:20) -- (B) arc (270:90:17.32) -- cycle;
    \end{scope}
    \draw[thick] (O1) circle (20);
    \draw[thick] (G) circle (17.32);
    \draw[thick] (A) -- (F);
    \draw[thick] (B) -- (D);
    \node[left] at (A) {$A$};
    \node[below] at (B) {$B$};
    \node[below right] at (C) {$C$};
    \node[above] at (D) {$D$};
    \node[below right] at (E) {$E$};
    \node[right] at (F) {$F$};
    \node[below left] at (G) {$G$};
\end{tikzpicture}
\end{minipage}
\begin{solution}
Let $O$ be the centre of circle $ABCD$. Since $AC = 30 + 10 = 40\text{ cm}$, the radius of circle $ABCD$ is $OA = OD = 20\text{ cm}$ and $OG = 30 - 20 = 10\text{ cm}$. \\
Since $BD$ and $EF$ are diameters of circle $BEDF$, $G$ is its centre and $AC \perp BD$. In right-angled $\triangle OGD$:
\[ GD = \sqrt{20^2 - 10^2} = 10\sqrt{3}\text{ cm} \quad \text{and} \quad \cos\angle DOG = \frac{10}{20} = \frac{1}{2} \implies \angle DOG = 60^\circ \]
Thus reflex $\angle BOD = 360^\circ - 120^\circ = 240^\circ$. The area of the shaded region is:
\begin{align*}
\text{Area} &= \text{Area}(\text{major sector } OBAD) + \text{Area}(\triangle OBD) - \text{Area}(\text{semi-circle } BED) \\
&= \pi(20)^2 \times \frac{240^\circ}{360^\circ} + \frac{1}{2}(20\sqrt{3})(10) - \frac{1}{2}\pi(10\sqrt{3})^2 \\
&= \frac{350}{3}\pi + 100\sqrt{3} \approx 540\text{ cm}^2 \quad \text{(D)}
\end{align*}
\end{solution}

% Q17
\question In the figure, $PQRS$ is a parallelogram. Let $X$ be a point lying on $PQ$ . Denote the point of intersection of $PR$ and $SX$ by $Y$ . If the area of $\triangle PXY$ and the area of the quadrilateral $QRYX$ are $32\text{ cm}^2$ and $58\text{ cm}^2$ respectively, then the area of $\triangle RSY$ is

\begin{minipage}{0.45\textwidth}
\begin{choices}
    \choice $40\text{ cm}^2$ .
    \correctchoice $50\text{ cm}^2$ .
    \choice $58\text{ cm}^2$ .
    \choice $72\text{ cm}^2$ .
\end{choices}
\end{minipage}
\begin{minipage}{0.52\textwidth}
\centering
\begin{tikzpicture}[scale=0.8]
    \coordinate (P) at (0,0);
    \coordinate (Q) at (5,0);
    \coordinate (S) at (1,3.2);
    \coordinate (R) at (6,3.2);
    \coordinate (X) at (4,0);
    \draw[thick] (P) -- (Q) -- (R) -- (S) -- cycle;
    \draw[thick] (P) -- (R);
    \draw[thick] (S) -- (X);
    \path[name path=PR] (P) -- (R);
    \path[name path=SX] (S) -- (X);
    \path[name intersections={of=PR and SX, by=Y}];
    \node[below left] at (P) {$P$};
    \node[below] at (X) {$X$};
    \node[below right] at (Q) {$Q$};
    \node[above left] at (S) {$S$};
    \node[above right] at (R) {$R$};
    \node[above] at (Y) {$Y$};
\end{tikzpicture}
\end{minipage}
\begin{solution}
$\text{Area of } \triangle PRS = \text{Area of } \triangle PQR = 32 + 58 = 90\text{ cm}^2$. \\
Let $\text{Area of } \triangle RSY = A\text{ cm}^2$. Then $\text{Area of } \triangle PSY = (90 - A)\text{ cm}^2$. \\
Since $\triangle PXY \sim \triangle RSY$:
\[ \left(\frac{90 - A}{A}\right)^2 = \left(\frac{PY}{RY}\right)^2 = \frac{32}{A} \implies (90 - A)^2 = 32A \implies A^2 - 212A + 8100 = 0 \]
Since $A < 90$, solving $(A - 50)(A - 162) = 0$ gives $A = 50\text{ cm}^2$ (B).
\end{solution}
