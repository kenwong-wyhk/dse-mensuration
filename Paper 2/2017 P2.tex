
% Q15
\question The base radius of a right circular cone is $2$ times the base radius of a right circular cylinder while the height of the circular cylinder is $3$ times the height of the circular cone. If the volume of the circular cone is $36\pi\text{ cm}^3$ , then the volume of the circular cylinder is
\begin{choices}
    \choice $27\pi\text{ cm}^3$ .
    \choice $48\pi\text{ cm}^3$ .
    \correctchoice $81\pi\text{ cm}^3$ .
    \choice $144\pi\text{ cm}^3$ .
\end{choices}
\begin{solution}
Let $r$ and $h$ be the base radius of the cylinder and the height of the cone respectively. \\
From the cone: $\frac{1}{3}\pi(2r)^2h = 36\pi \implies \pi r^2h = 27\pi$. \\
Volume of the cylinder $= \pi r^2(3h) = 3(27\pi) = 81\pi\text{ cm}^3$ (C).
\end{solution}

% Q16
\question In the figure, $ABCD$ and $BEDF$ are parallelograms. $E$ is a point lying on $BC$ such that $BE : EC = 2 : 3$ . $AC$ cuts $BF$ and $DE$ at $G$ and $H$ respectively. If the area of $\triangle ABG$ is $135\text{ cm}^2$ , then the area of the quadrilateral $DFGH$ is

\begin{minipage}{0.45\textwidth}
\begin{choices}
    \choice $60\text{ cm}^2$ .
    \choice $81\text{ cm}^2$ .
    \choice $90\text{ cm}^2$ .
    \correctchoice $144\text{ cm}^2$ .
\end{choices}
\end{minipage}
\begin{minipage}{0.52\textwidth}
\centering
\begin{tikzpicture}[scale=0.8]
    \coordinate (C) at (0,0);
    \coordinate (E) at (2.7,0);
    \coordinate (B) at (4.5,0);
    \coordinate (D) at (1.5,3.0);
    \coordinate (F) at (3.3,3.0);
    \coordinate (A) at (6.0,3.0);
    \path[name path=AC] (A) -- (C);
    \path[name path=BF] (B) -- (F);
    \path[name path=DE] (D) -- (E);
    \path[name intersections={of=AC and BF, by=G}];
    \path[name intersections={of=AC and DE, by=H}];
    \draw[thick] (A) -- (B) -- (C) -- (D) -- cycle;
    \draw[thick] (A) -- (C);
    \draw[thick] (B) -- (F);
    \draw[thick] (D) -- (E);
    \node[above right] at (A) {$A$};
    \node[below] at (B) {$B$};
    \node[below left] at (C) {$C$};
    \node[above left] at (D) {$D$};
    \node[below] at (E) {$E$};
    \node[above] at (F) {$F$};
    \node[above=1pt, left=1pt] at (G) {$G$};
    \node[above=1pt, left=1pt] at (H) {$H$};
\end{tikzpicture}
\end{minipage}
\begin{solution}
Since $DF = BE = \frac{2}{5}BC = \frac{2}{5}AD$, we have $AF : FD = 3 : 2$ and $AF : BC = 3 : 5$. \\
Since $\triangle AFG \sim \triangle CBG$, $FG : BG = AF : BC = 3 : 5$, so $\text{Area}(\triangle AFG) = \frac{3}{5}(135) = 81\text{ cm}^2$. \\
Since $FG$ // $DH$, $\triangle AFG \sim \triangle ADH$ with $AF : AD = 3 : 5$, so:
\[ \text{Area}(DFGH) = 81 \times \left[\left(\frac{5}{3}\right)^2 - 1\right] = 81 \times \frac{16}{9} = 144\text{ cm}^2 \quad \text{(D)} \]
\end{solution}
