
% Q15
\question The perimeter of the sector $OMN$ is $12\pi\text{ cm}$ , where $O$ is the centre of the sector $OMN$ . If the radius of the sector $OMN$ is $3\pi\text{ cm}$ , which of the following are true?
\begin{itemize}
    \item[I.] The area of the sector $OMN$ is $9\pi^2\text{ cm}^2$ .
    \item[II.] The perimeter of $\triangle OMN$ is less than $35\text{ cm}$ .
    \item[III.] The angle of the sector $OMN$ is greater than $100^\circ$ .
\end{itemize}
\begin{choices}
    \choice I and II only
    \choice I and III only
    \choice II and III only
    \correctchoice I, II and III
\end{choices}
\begin{solution}
Let $r = 3\pi\text{ cm}$. The arc length $\widehat{MN} = 12\pi - 2(3\pi) = 6\pi\text{ cm}$. \mk{1M}
\begin{itemize}
    \item I is true: Area of sector $= \frac{1}{2}r\widehat{MN} = \frac{1}{2}(3\pi)(6\pi) = 9\pi^2\text{ cm}^2$.
    \item III is true: $\angle MON = \dfrac{6\pi}{3\pi}\text{ rad} = 2\text{ rad} = \dfrac{360^\circ}{\pi} \approx 114.6^\circ > 100^\circ$.
    \item II is true: Perimeter of $\triangle OMN = 6\pi + 2(3\pi)\sin(1\text{ rad}) \approx 34.71\text{ cm} < 35\text{ cm}$. \mk{1A}
\end{itemize}
Thus, the answer is D.
\end{solution}

% Q16
\question The height and the total surface area of a solid right circular cylinder are $35\text{ cm}$ and $492\pi\text{ cm}^2$ respectively. If the radius of a solid sphere is equal to the base radius of the circular cylinder, then the volume of the sphere is
\begin{choices}
    \choice $144\pi\text{ cm}^3$ .
    \correctchoice $288\pi\text{ cm}^3$ .
    \choice $576\pi\text{ cm}^3$ .
    \choice $864\pi\text{ cm}^3$ .
\end{choices}
\begin{solution}
Let $r\text{ cm}$ be the base radius of the cylinder.
\begin{align*}
2\pi r^2 + 2\pi r(35) &= 492\pi \mk{1M} \\
r^2 + 35r - 246 &= 0 \\
(r - 6)(r + 41) &= 0 \implies r = 6
\end{align*}
Volume of the sphere $= \dfrac{4}{3}\pi(6)^3 = 288\pi\text{ cm}^3$ (B). \mk{1A}
\end{solution}

% Q17
\question In the figure, $ABCD$ is a parallelogram. Let $E$ , $F$ and $G$ be points lying on $AB$ , $AD$ and $CD$ respectively such that $BE = 3AE$ , $2DF = 3AF$ and $DG = 2CG$ . It is given that $BC$ produced and $EG$ produced meet at the point $H$ . If the area of $\triangle CGH$ is $16\text{ cm}^2$ , then the area of the quadrilateral $DFEG$ is

\begin{minipage}{0.45\textwidth}
\begin{choices}
    \choice $46\text{ cm}^2$ .
    \correctchoice $49\text{ cm}^2$ .
    \choice $105\text{ cm}^2$ .
    \choice $115\text{ cm}^2$ .
\end{choices}
\end{minipage}
\begin{minipage}{0.52\textwidth}
\centering
\begin{tikzpicture}[scale=0.75]
    \coordinate (A) at (0,0);
    \coordinate (B) at (4.8,0);
    \coordinate (D) at (0.8,3.2);
    \coordinate (C) at (5.6,3.2);
    \coordinate (E) at (1.2,0);
    \coordinate (F) at ($(A)!0.4!(D)$);
    \coordinate (G) at ($(D)!0.667!(C)$);
    \coordinate (H) at (6.24,5.76);
    \draw[thick] (D) -- (A) -- (B) -- (C) -- cycle;
    \draw[thick] (C) -- (H) -- (E) -- (F);
    \node[below left] at (A) {$A$};
    \node[below] at (E) {$E$};
    \node[below] at (B) {$B$};
    \node[left] at (F) {$F$};
    \node[left] at (D) {$D$};
    \node[above] at (G) {$G$};
    \node[right] at (C) {$C$};
    \node[above right] at (H) {$H$};
\end{tikzpicture}
\end{minipage}
\begin{solution}
Since $CG = \frac{1}{3}CD$ and $BE = \frac{3}{4}AB = \frac{3}{4}CD$, we have $CG : BE = 4 : 9$. \\
Since $\triangle CGH \sim \triangle BEH$:
\[ \text{Area of } \triangle BEH = 16 \times \left(\frac{9}{4}\right)^2 = 81\text{ cm}^2 \implies \text{Area of } BCGE = 81 - 16 = 65\text{ cm}^2 \mk{1M} \]
Since trapezium $BCGE$ occupies $\frac{1}{2}\left(\frac{1}{3} + \frac{3}{4}\right) = \frac{13}{24}$ of the area of $ABCD$:
\[ \text{Area of } ABCD = 65 \times \frac{24}{13} = 120\text{ cm}^2 \quad \text{and} \quad \text{Area of } \triangle AEF = \frac{1}{2}\left(\frac{1}{4}\right)\left(\frac{2}{5}\right)(120) = 6\text{ cm}^2 \]
Thus, $\text{Area of } DFEG = 120 - 65 - 6 = 49\text{ cm}^2$ (B). \mk{1A}
\end{solution}
